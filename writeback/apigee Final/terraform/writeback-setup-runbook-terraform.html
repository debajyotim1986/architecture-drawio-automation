T4How every stack is runscripts/tf.sh

Platform
#!/usr/bin/env bash
# Usage: scripts/tf.sh <env> <stack> <init|plan|apply|destroy-plan|output> [extra terraform args]
#   env   : int | pre | prd
#   stack : 10-foundation | 15-entra | 20-workloads | 40-tenant-<key>
# Always plans to a file and applies exactly that plan.
set -euo pipefail

ENV="${1:?env}"; STACK="${2:?stack}"; CMD="${3:?command}"; shift 3 || true
ROOT="$(cd "$(dirname "$0")/.." && pwd)"
case "$STACK" in
  40-tenant-*) DIR="$ROOT/stacks/40-tenant-writer" ;;
  *)           DIR="$ROOT/stacks/$STACK" ;;
esac
VARS="$ROOT/envs/$ENV/$STACK.tfvars"
BACKEND="$ROOT/envs/$ENV/backend-$STACK.hcl"
PLAN="$ROOT/.plans/$ENV-$STACK.tfplan"
mkdir -p "$ROOT/.plans"
export TF_DATA_DIR="$ROOT/.terraform-data/$ENV-$STACK"   # keeps one .terraform per env+stack

tf() { terraform -chdir="$DIR" "$@"; }

case "$CMD" in
  init)         tf init -reconfigure -backend-config="$BACKEND" "$@" ;;
  plan)         tf init -reconfigure -backend-config="$BACKEND" -input=false >/dev/null
                tf plan -input=false -var-file="$VARS" -out="$PLAN" "$@" ;;
  apply)        tf apply -input=false "$PLAN" ;;
  destroy-plan) tf plan -destroy -input=false -var-file="$VARS" -out="$PLAN" "$@" ;;
  output)       tf output "$@" ;;
  *) echo "unknown command $CMD"; exit 2 ;;
esac

Plan is always written to a file, reviewed (pull request comment / pipeline approval), and exactly that file is applied. Each env + stack has its own state prefix writeback/<env>/<stack> and its own .terraform folder.

# C1 / C4-1 : Workload Identity must already be on for the existing Looker cluster.
check "looker_cluster_workload_identity" {
  assert {
    condition = (
      length(data.google_container_cluster.looker.workload_identity_config) > 0 &&
      data.google_container_cluster.looker.workload_identity_config[0].workload_pool == local.looker_pool_id
    )
    error_message = "Looker cluster has no workload pool ${local.looker_pool_id}. Run gcloud runbook step C1 once (cluster is not managed by Terraform)."
  }
}

# C2 / C4-2 : every Looker node pool must use the GKE metadata server.
check "looker_node_pools_gke_metadata" {
  assert {
    condition = alltrue([
      for np in data.google_container_cluster.looker.node_pool :
      try(np.node_config[0].workload_metadata_config[0].mode, "") == "GKE_METADATA"
    ])
    error_message = "A Looker node pool is not on GKE_METADATA. Run gcloud runbook step C2 for that pool (rolls the nodes – plan a window)."
  }
}

# Secret Manager add-on on the Looker cluster is needed to mount action-hub TLS from Secret Manager.
check "looker_cluster_secret_manager_addon" {
  assert {
    condition     = try(data.google_container_cluster.looker.secret_manager_config[0].enabled, false)
    error_message = "Secret Manager add-on is off on the Looker cluster. Enable it once: gcloud container clusters update ${var.looker_cluster} --enable-secret-manager --location=${var.region} --project=${var.looker_project}"
  }
}

A failing check prints the warning in the plan but does not stop it. Treat any check warning as a stop: run the one gcloud command it prints (these are the gcloud runbook steps C1, C2 and the add-on switch) through the Looker cluster's own change process, then plan again.

"""
P1–P5 · Looker configuration as code (there is no official Terraform provider for Looker).
Idempotent: safe to run again; it only creates what is missing.

Run by a Looker admin, from a host that can read looker-action-secret:
    pip install looker-sdk google-cloud-secret-manager
    export LOOKERSDK_BASE_URL=https://looker-int.internal   LOOKERSDK_CLIENT_ID=...   LOOKERSDK_CLIENT_SECRET=...
    python configure_looker.py --env int --action-hub-url https://action-hub.looker-actions.svc.cluster.local \
        --secret projects/looker-prj-int/secrets/looker-action-secret/versions/latest --model writeback_model

The Looker API credentials come from the admin's environment (or a Secret Manager–backed
CI variable) — never from a file in the repository. The action secret is read from
Secret Manager and sent straight to Looker; it is never printed.
"""
import argparse

import looker_sdk
from looker_sdk import models40 as m
from google.cloud import secretmanager

GROUP = "Writeback Users"
ATTR = "writeback_allowed"
PERMISSION_SET = "writeback-send"
MODEL_SET = "writeback-models"
ROLE = "Writeback User"
# send_to_integration needs access_data; Looker rejects the set if a dependency is missing
PERMISSIONS = ["access_data", "see_looks", "see_user_dashboards", "send_to_integration"]


def one(items, **match):
    for it in items or []:
        if all(getattr(it, k) == v for k, v in match.items()):
            return it
    return None


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--env", required=True)
    ap.add_argument("--action-hub-url", required=True)
    ap.add_argument("--secret", required=True, help="Secret Manager version name of looker-action-secret")
    ap.add_argument("--model", required=True, help="LookML model that carries the write-back explores")
    args = ap.parse_args()

    sdk = looker_sdk.init40()

    # P1 — manual group
    group = one(sdk.search_groups(name=GROUP), name=GROUP) or sdk.create_group(m.WriteGroup(name=GROUP))
    print(f"P1 group      {GROUP} id={group.id}")

    # P2 — hidden user attribute, default 'no'
    attr = one(sdk.all_user_attributes(), name=ATTR) or sdk.create_user_attribute(m.WriteUserAttribute(
        name=ATTR, label="Write-back allowed", type="string", default_value="no",
        value_is_hidden=True, user_can_view=False, user_can_edit=False))
    print(f"P2 attribute  {ATTR} id={attr.id}")

    # P3 — group value yes (never on individual users)
    sdk.set_user_attribute_group_values(attr.id, [m.UserAttributeGroupValue(group_id=group.id, value="yes")])
    print(f"P3 value      {GROUP} → {ATTR}=yes")

    # P4 — role with send_to_integration on the write-back model, given to the group
    pset = one(sdk.all_permission_sets(), name=PERMISSION_SET) or sdk.create_permission_set(
        m.WritePermissionSet(name=PERMISSION_SET, permissions=PERMISSIONS))
    mset = one(sdk.all_model_sets(), name=MODEL_SET) or sdk.create_model_set(
        m.WriteModelSet(name=MODEL_SET, models=[args.model]))
    role = one(sdk.all_roles(), name=ROLE) or sdk.create_role(
        m.WriteRole(name=ROLE, permission_set_id=pset.id, model_set_id=mset.id))
    current = {g.id for g in sdk.role_groups(role.id)}
    sdk.set_role_groups(role.id, sorted(current | {group.id}))
    print(f"P4 role       {ROLE} id={role.id} → {GROUP}")

    # P5 — register the Action Hub with the shared secret, then enable the write-back action
    token = secretmanager.SecretManagerServiceClient().access_secret_version(
        name=args.secret).payload.data.decode()
    hub = one(sdk.all_integration_hubs(), url=args.action_hub_url)
    if hub:
        sdk.update_integration_hub(hub.id, m.WriteIntegrationHub(url=args.action_hub_url, authorization_token=token))
    else:
        hub = sdk.create_integration_hub(m.WriteIntegrationHub(url=args.action_hub_url, authorization_token=token))
    del token
    health = sdk.get_integration_hub_health(hub.id)  # confirms Looker can reach the Action Hub
    print(f"P5 hub        id={hub.id} health={getattr(health, 'success', health)}")
    for integ in sdk.all_integrations(integration_hub_id=hub.id):
        if integ.label and "write back" in integ.label.lower():
            sdk.update_integration(integ.id, m.WriteIntegration(enabled=True))
            print(f"P5 action     {integ.label} enabled on hub {hub.id}")
    print("P6 (LookML tags) and P7 (add pilot users to the group) stay in the normal change process.")


if __name__ == "__main__":
